}

# ---- Option 1 : Generer un manifeste -----------------------------------------

function Invoke-GenererManifeste {
    Show-Header
    Write-Host "  GENERATION DU MANIFESTE SHA-256" -ForegroundColor Green
    Write-Host "  ----------------------------------------------------------------" -ForegroundColor DarkGray
    Write-Host ""

    # Saisie du chemin source
    Write-Host "  Chemin source (disque ou dossier a hacher)" -ForegroundColor White
    Write-Host "  Exemple : I:\"  -ForegroundColor DarkGray
    $source = (Read-Host "  > ").Trim()

    if (-not (Test-Path $source)) {
        Write-Host ""
        Write-Host "  ERREUR : '$source' introuvable." -ForegroundColor Red
        Write-Host "  Verifiez que le disque est connecte et deverrouille." -ForegroundColor Red
        Pause-Retour
        return
    }

    # Saisie du chemin de sortie
    Write-Host ""
    Write-Host "  Chemin complet du fichier manifeste a creer" -ForegroundColor White
    Write-Host "  Exemple : D:\manifeste_SHA256.txt" -ForegroundColor DarkGray
    $sortie = (Read-Host "  > ").Trim()

    if ([string]::IsNullOrWhiteSpace($sortie)) {
        Write-Host ""
        Write-Host "  ERREUR : Chemin de sortie invalide." -ForegroundColor Red
        Pause-Retour
        return
    }

    # Creer le dossier de sortie si necessaire
    $dossierSortie = Split-Path -Parent $sortie
    if ($dossierSortie -and -not (Test-Path $dossierSortie)) {
        New-Item -ItemType Directory -Path $dossierSortie -Force | Out-Null
    }

    Write-Host ""
    Write-Host "  Inventaire des fichiers..." -ForegroundColor Yellow
    $fichiers = Get-ChildItem -Path $source -Recurse -File -ErrorAction SilentlyContinue
    $total    = $fichiers.Count

    if ($total -eq 0) {
        Write-Host "  ERREUR : Aucun fichier trouve dans '$source'." -ForegroundColor Red
        Pause-Retour
        return
    }

    Write-Host "  $total fichier(s) trouve(s). Debut du hachage..." -ForegroundColor Green
    Write-Host ""

    $compteur = 0
    $debut    = Get-Date

    $fichiers | ForEach-Object {
        $compteur++
        $hash = (certutil -hashfile $_.FullName SHA256)[1]
        "$hash  $($_.FullName)"

        $pct    = [int](($compteur / $total) * 100)
        $ecoule = (Get-Date) - $debut
        if ($compteur -gt 1) {
            $restantSec   = [int](($total - $compteur) * ($ecoule.TotalSeconds / $compteur))
            $tempsRestant = "{0}m {1}s" -f [int]($restantSec / 60), ($restantSec % 60)
        } else {
            $tempsRestant = "calcul en cours..."
        }

        Write-Progress `
            -Activity        "Generation du manifeste SHA-256" `
            -Status          "Fichier $compteur / $total  —  Reste : $tempsRestant" `
            -CurrentOperation $_.FullName `
            -PercentComplete $pct

    } | Out-File $sortie -Encoding UTF8

    Write-Progress -Activity "Generation du manifeste SHA-256" -Completed

    # Rapport
    $duree = (Get-Date) - $debut
    Write-Host "  ----------------------------------------------------------------" -ForegroundColor DarkGray
    Write-Host "  TERMINE" -ForegroundColor Green
    Write-Host "  Fichiers traites : $total" -ForegroundColor White
    Write-Host "  Duree            : $([int]$duree.TotalMinutes)m $($duree.Seconds)s" -ForegroundColor White
    Write-Host "  Manifeste        : $sortie" -ForegroundColor White
    Write-Host ""

    # Empreinte du manifeste — a transmettre au destinataire
    $hashManifeste = (certutil -hashfile $sortie SHA256)[1].Trim()
    Write-Host "  ----------------------------------------------------------------" -ForegroundColor DarkGray
    Write-Host "  EMPREINTE SHA-256 DU MANIFESTE (a transmettre au destinataire)" -ForegroundColor Cyan
    Write-Host ""
    Write-Host "  $hashManifeste" -ForegroundColor Yellow
    Write-Host ""
    Write-Host "  Transmettez cette empreinte via canal securise" -ForegroundColor DarkGray
    Write-Host "  (Signal, verbal) — separement du disque dur." -ForegroundColor DarkGray
    Write-Host "  ----------------------------------------------------------------" -ForegroundColor DarkGray

    Pause-Retour
}

# ---- Option 2 : Verifier un manifeste ----------------------------------------

function Invoke-VerifierManifeste {
    Show-Header
    Write-Host "  VERIFICATION D'INTEGRITE DU DISQUE" -ForegroundColor Yellow
    Write-Host "  ----------------------------------------------------------------" -ForegroundColor DarkGray
    Write-Host ""

    # Saisie du disque a verifier
    Write-Host "  Chemin du disque ou dossier a verifier" -ForegroundColor White
    Write-Host "  Exemple : E:\" -ForegroundColor DarkGray
    $disque = (Read-Host "  > ").Trim()

    if (-not (Test-Path $disque)) {
        Write-Host ""
        Write-Host "  ERREUR : '$disque' introuvable." -ForegroundColor Red
        Write-Host "  Verifiez que le disque est connecte et deverrouille par BitLocker." -ForegroundColor Red
        Pause-Retour
        return
    }

    # Saisie du chemin du manifeste de reference
    Write-Host ""
    Write-Host "  Chemin du manifeste de reference (recu de l'expediteur)" -ForegroundColor White
    Write-Host "  Exemple : C:\manifeste_SHA256.txt" -ForegroundColor DarkGray
    $manifestRef = (Read-Host "  > ").Trim()

    if (-not (Test-Path $manifestRef)) {
        Write-Host ""
        Write-Host "  ERREUR : '$manifestRef' introuvable." -ForegroundColor Red
        Pause-Retour
        return
    }

    # Saisie de l'empreinte du manifeste transmise par l'expediteur
    Write-Host ""
    Write-Host "  Empreinte SHA-256 du manifeste (transmise par l'expediteur)" -ForegroundColor White
    Write-Host "  Collez la valeur recue via canal securise :" -ForegroundColor DarkGray
    $empreinteRecue = (Read-Host "  > ").Trim()

    # Calcul et comparaison automatique de l'empreinte
    Write-Host ""
    Write-Host "  Verification de l'empreinte du manifeste..." -ForegroundColor Yellow
    $empreinteCalculee = (certutil -hashfile $manifestRef SHA256)[1].Trim()

    if ($empreinteCalculee -ne $empreinteRecue) {
        Write-Host ""
        Write-Host "  ============================================" -ForegroundColor Red
        Write-Host "   ALERTE : LE MANIFESTE A ETE ALTERE" -ForegroundColor Red
        Write-Host "  ============================================" -ForegroundColor Red
        Write-Host ""
        Write-Host "  Empreinte recue    : $empreinteRecue" -ForegroundColor Red
        Write-Host "  Empreinte calculee : $empreinteCalculee" -ForegroundColor Red
        Write-Host ""
        Write-Host "  Le fichier manifeste ne correspond pas a l'empreinte" -ForegroundColor Red
        Write-Host "  transmise par l'expediteur. Ne pas utiliser ces donnees." -ForegroundColor Red
        Pause-Retour
        return
    }

    Write-Host "  Empreinte du manifeste : VALIDE" -ForegroundColor Green
    Write-Host ""

    # Chargement du manifeste de reference
    Write-Host "  Chargement du manifeste de reference..." -ForegroundColor Yellow
    $lignesRef = Get-Content $manifestRef -Encoding UTF8 | Where-Object { $_ -match '\S' }

    $lookupRef = @{}
    foreach ($ligne in $lignesRef) {
        $parts = $ligne -split '\s+', 2
        if ($parts.Count -eq 2) {
            $hash          = $parts[0].Trim()
            $cheminRelatif = ($parts[1].Trim()) -replace '^[A-Za-z]:\\', ''
            $lookupRef[$cheminRelatif] = $hash
        }
    }

    $totalRef = $lookupRef.Count
    Write-Host "  $totalRef fichier(s) dans le manifeste de reference." -ForegroundColor Green

    # Inventaire du disque
    Write-Host ""
    Write-Host "  Inventaire du disque recu..." -ForegroundColor Yellow
    $fichiers    = Get-ChildItem -Path $disque -Recurse -File -ErrorAction SilentlyContinue
    $totalDisque = $fichiers.Count
    Write-Host "  $totalDisque fichier(s) present(s) sur le disque." -ForegroundColor Green
    Write-Host ""

    # Recalcul et comparaison
    Write-Host "  Recalcul SHA-256 et comparaison..." -ForegroundColor Yellow
    Write-Host ""

    $compteur  = 0
    $okCount   = 0
    $erreurs   = [System.Collections.Generic.List[string]]::new()
    $manquants = [System.Collections.Generic.List[string]]::new()
    $debut     = Get-Date

    foreach ($fichier in $fichiers) {
        $compteur++

        $hashCalcule   = (certutil -hashfile $fichier.FullName SHA256)[1].Trim()
        $cheminRelatif = $fichier.FullName -replace '^[A-Za-z]:\\', ''

        if ($lookupRef.ContainsKey($cheminRelatif)) {
            if ($hashCalcule -eq $lookupRef[$cheminRelatif]) {
                $okCount++
            } else {
                $erreurs.Add("  [HASH DIFFERENT] $($fichier.FullName)")
            }
        } else {
            $manquants.Add("  [ABSENT DU MANIFESTE] $($fichier.FullName)")
        }

        $pct    = [int](($compteur / [math]::Max($totalDisque, 1)) * 100)
        $ecoule = (Get-Date) - $debut
        if ($compteur -gt 1) {
            $restantSec   = [int](($totalDisque - $compteur) * ($ecoule.TotalSeconds / $compteur))
            $tempsRestant = "{0}m {1}s" -f [int]($restantSec / 60), ($restantSec % 60)
        } else {
            $tempsRestant = "calcul en cours..."
        }

        Write-Progress `
            -Activity        "Verification SHA-256" `
            -Status          "Fichier $compteur / $totalDisque  —  OK: $okCount  Erreurs: $($erreurs.Count)  Reste: $tempsRestant" `
            -CurrentOperation $fichier.FullName `
            -PercentComplete $pct
    }

    Write-Progress -Activity "Verification SHA-256" -Completed

    # Fichiers dans le manifeste mais absents du disque
    $absents = [System.Collections.Generic.List[string]]::new()
    foreach ($cheminRelatif in $lookupRef.Keys) {
        $cheminComplet = $disque.TrimEnd('\') + '\' + $cheminRelatif
        if (-not (Test-Path $cheminComplet)) {
            $absents.Add("  [FICHIER MANQUANT] $cheminComplet")
        }
    }

    # Rapport final
    $duree          = (Get-Date) - $debut
    $totalProblemes = $erreurs.Count + $manquants.Count + $absents.Count

    Write-Host ""
    if ($totalProblemes -eq 0) {
        Write-Host "  ============================================" -ForegroundColor Green
        Write-Host "   INTEGRITE CONFIRMEE - AUCUNE ANOMALIE" -ForegroundColor Green
        Write-Host "  ============================================" -ForegroundColor Green
        Write-Host ""
        Write-Host "  Fichiers verifies : $okCount / $totalRef" -ForegroundColor White
        Write-Host "  Anomalies         : 0" -ForegroundColor Green
        Write-Host "  Duree             : $([int]$duree.TotalMinutes)m $($duree.Seconds)s" -ForegroundColor White
        Write-Host ""
        Write-Host "  Le transfert est integre." -ForegroundColor Green
    } else {
        Write-Host "  ============================================" -ForegroundColor Red
        Write-Host "   ANOMALIES DETECTEES" -ForegroundColor Red
        Write-Host "  ============================================" -ForegroundColor Red
        Write-Host ""
        Write-Host "  Fichiers OK             : $okCount / $totalRef" -ForegroundColor White
        Write-Host "  Hash differents         : $($erreurs.Count)" -ForegroundColor Red
        Write-Host "  Absents du manifeste    : $($manquants.Count)" -ForegroundColor Yellow
        Write-Host "  Manquants sur le disque : $($absents.Count)" -ForegroundColor Yellow
        Write-Host "  Duree                   : $([int]$duree.TotalMinutes)m $($duree.Seconds)s" -ForegroundColor White
        Write-Host ""

        if ($erreurs.Count -gt 0) {
            Write-Host "  --- Hash differents ---" -ForegroundColor Red
            $erreurs | Select-Object -First 10 | ForEach-Object { Write-Host $_ -ForegroundColor Red }
            if ($erreurs.Count -gt 10) { Write-Host "  ... et $($erreurs.Count - 10) autre(s)." -ForegroundColor Red }
            Write-Host ""
        }
        if ($absents.Count -gt 0) {
            Write-Host "  --- Fichiers manquants sur le disque ---" -ForegroundColor Yellow
            $absents | Select-Object -First 10 | ForEach-Object { Write-Host $_ -ForegroundColor Yellow }
            Write-Host ""
        }
        if ($manquants.Count -gt 0) {
            Write-Host "  --- Fichiers non repertories dans le manifeste ---" -ForegroundColor Yellow
            $manquants | Select-Object -First 10 | ForEach-Object { Write-Host $_ -ForegroundColor Yellow }
            Write-Host ""
        }

        # Export du rapport d'anomalies
        $dossier     = Split-Path -Parent $manifestRef
        $rapportPath = Join-Path $dossier "rapport_anomalies_$(Get-Date -Format 'yyyyMMdd_HHmmss').txt"
        ($erreurs + $absents + $manquants) | Out-File $rapportPath -Encoding UTF8
        Write-Host "  Rapport exporte : $rapportPath" -ForegroundColor Yellow
    }

    Pause-Retour
}
